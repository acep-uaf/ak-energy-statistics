pce_outliers_log = FileAttachment("data/l2/logs/l2_pce_imputation_options.csv")
  .csv()
  .then(data => {
    return data.map(d => {
      let cleanedRow = {};

      for (let key in d) {
        let val = d[key];

        if (val === "" || val === "NA" || val === null || val === undefined) {
          cleanedRow[key] = null;
        }
        else if (key === "date" || isNaN(val) || isNaN(Number(val))) {
          cleanedRow[key] = val;
        }
        else {
          cleanedRow[key] = Number(val);
        }
      }

      if (!cleanedRow.status) {
        cleanedRow.status = "Carry forward";
      }

      return cleanedRow;
    })
  })

viewof outlierTableData = {
  const initialData = pce_outliers_log.filter(d =>
    d.sales_reporting_name === selectedCommunity &&
    d.column_tested === selectedColumn
  );

  const container = html`<div style="margin-bottom: 20px;"></div>`;

  const table = new Tabulator(container, {
    data: JSON.parse(JSON.stringify(initialData)),
    layout: "fitDataFill",
    columns: [
      { title: "PCE ID", field: "project_code" },
      { title: "Name", field: "sales_reporting_name" },
      { title: "Year", field: "calendar_year" },
      { title: "Month", field: "calendar_month" },
      { title: "Column", field: "column_tested"},
      { title: "Orig Value", field: "raw_value" },
      { title: "Carry Fwd", field: "carry_forward" },
      { title: "Annual Avg", field: "annual_average" },
      { title: "Pre/Post Avg", field: "avg_preceding_proceeding" },
      { title: "Same Month Avg", field: "avg_same_month_other_years" },
      {
        title: "Override Action",
        field: "status",
        editor: "list",
        editorParams: {
          values: [
            "Value out of bounds, but original value kept",
            "Carry forward",
            "Annual average",
            "Average of preceding and proceeding months",
            "Average of months from other years",
            "Manual override: expert opinion"
          ]
        }
      }
    ]
  });


  const updateData = () => {
    container.value = table.getData();
    container.dispatchEvent(new CustomEvent("input"));
  };

  table.on("cellEdited", updateData);
  table.on("tableBuilt", updateData);

  container.value = initialData;
  return container;
}
