pce_outliers_log = FileAttachment("data/l2/logs/l2_pce_outliers_log.csv")
  .csv()
  .then(data => {
    return data.map(d => {
      let cleanedRow = {};

      for (let key in d) {
        let val = d[key];

        if (val === "" || val === "NA" || val === null || val === undefined) {
          cleanedRow[key] = null;
        }
        else if (key === "date" || isNaN(val) || isNaN(Number(val))) {
          cleanedRow[key] = val;
        }
        else {
          cleanedRow[key] = Number(val);
        }
      }

      return cleanedRow;
    })
  })

pce = FileAttachment("data/l1/consolidated/l1_pce.csv")
  .csv()
  .then(data => {
    return data.map(d => {
      let cleanedRow = {};

      for (let key in d) {
        let val = d[key];

        if (val === "" || val === "NA" || val === null || val === undefined) {
          cleanedRow[key] = null;
        }
        else if (key === "date" || isNaN(val) || isNaN(Number(val))) {
          cleanedRow[key] = val;
        }
        else {
          cleanedRow[key] = Number(val);
        }
      }

      return cleanedRow;
    })
    .sort((a, b) => new Date(a.date) - new Date(b.date));
  })
communities = [...new Set(pce_outliers_log.map(d => d.sales_reporting_name))]

viewof selectedCommunity = Inputs.select(
  communities,
  { label: "Community:", value: communities[0] }
)

availableColumns = {
  const subset = pce_outliers_log.filter(d => d.sales_reporting_name === selectedCommunity);
  return [...new Set(subset.map(d => d.column_tested))];
}

viewof selectedColumn = Inputs.select(
  availableColumns,
  { label: "Column:", value: availableColumns[0] }
)
filteredFullSeries = pce.filter(d =>
  d.sales_reporting_name === selectedCommunity
)

filteredOutliers = pce_outliers_log.filter(d =>
  d.sales_reporting_name === selectedCommunity &&
  d.column_tested === selectedColumn
)
// plot lines, dots, outlier dots

Plot.plot({
  height: 500,
  width: 1600,
  marginLeft: 60,
  grid: false,
  x: {
    type: "utc",
    label: "Date",
    ticks: 20,
    tickFormat: "%Y"
  },
  y: {
    label: selectedColumn !== "All" ? selectedColumn : "Value",
    grid: true
  },
  marks: [
    // timeseries line
    Plot.line(filteredFullSeries, {
      x: "date",
      y: d => d[selectedColumn],
      stroke: "#ccc",
      strokeWidth: 1.5
    }),

    // timeseries dots
    Plot.dot(filteredFullSeries, {
      x: "date",
      y: d => d[selectedColumn],
      fill: "#888888",
      r: 3
    }),

    // outlier dots
    Plot.dot(filteredOutliers, {
      x: "date",
      y: d => d.raw_value ?? d.value,
      fill: "red",
      stroke: "black",
      strokeWidth: 1,
      r: 6
    }),


    Plot.tip(filteredFullSeries, Plot.pointer({
        x: "date",
        y: d => d[selectedColumn],
        channels: {
            "Community": "sales_reporting_name",
            "Year": "calendar_year",
            "Month": "calendar_month",
            [selectedColumn]: d => d[selectedColumn]
        },
        format: {
            x: false,
            y: false
        }
    }))
  ]
})
tableContainer = {

            "Manual override: expert opinion"
            ] }

    ]
  });

  window.currentTable = table;
