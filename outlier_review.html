idb = require("idb@8.0.0/build/umd.js")
db = idb.openDB("OutlierReviewDB", 1, {
  upgrade(db) {
    if (!db.objectStoreNames.contains("reviews")) {
      db.createObjectStore("reviews", { keyPath: "identifier" });
    }
  }
})

// Helper to save or update an edited record
saveReview = async (record) => {
  if (!record || !record.identifier) return;
  const database = await db;
  await database.put("reviews", record);
}

// Helper to retrieve all saved records for CSV download
getAllSavedReviews = async () => {
  const database = await db;
  return await database.getAll("reviews");
}

// Helper to get saved state for a specific set of records
getSavedReviewsMap = async () => {
  const database = await db;
  const all = await database.getAll("reviews");
  return new Map(all.map(item => [item.identifier, item.decision]));
}

// Trigger to tell our dropdowns when to recalculate checkmarks

// Reactive map of all decisiones across the entire dataset
globalDecisionMap = {
  idbTrigger;
  
  const map = new Map();
  // Base decisiones from original dataset
  pce_outliers_log.forEach(d => {
    if (d.decision) map.set(d.identifier, d.decision);
  });
  
  // Overrides from IDB
  const saved = await getAllSavedReviews();
  saved.forEach(d => {
    if (d.decision) map.set(d.identifier, d.decision);
  });
  
  return map;
}

pce_outliers_log = FileAttachment("data/l2/logs/l2_pce_imputation_options.csv")
  .csv()
  .then(data => {
    return data.map(d => {
      let cleanedRow = {};

      for (let key in d) {
        let val = d[key];

        if (val === "" || val === "NA" || val === null || val === undefined) {
          cleanedRow[key] = null;
        }
        else if (key === "date" || isNaN(val) || isNaN(Number(val))) {
          cleanedRow[key] = val;
        }
        else {
          cleanedRow[key] = Number(val);
        }
      }

      if (!cleanedRow.decision) {
        cleanedRow.decision = null;
      }

      return cleanedRow;
    })
  })

viewof outlierTableData = {
  hideDecided; 

  const initialData = pce_outliers_log.filter(d =>
    d.sales_reporting_name === currentComm &&
    d.column === currentCol
  );

  const savedMap = await getSavedReviewsMap();
  const hydratedData = initialData.map(row => {
    if (savedMap.has(row.identifier)) {
      return { 
        ...row, 
        decision: savedMap.get(row.identifier),
        manual_override: savedMap.has(row.identifier) ? undefined : row.manual_override
      };
    }
    return row;
  });
  
  const allSaved = await getAllSavedReviews();
  const savedDict = Object.fromEntries(allSaved.map(item => [item.identifier, item]));
  
  const fullyHydratedData = initialData.map(row => {
    if (savedDict[row.identifier]) {
      return { ...row, ...savedDict[row.identifier] };
    }
    return row;
  });

  const container = html`<div style="margin-bottom: 20px; width: 100%;"></div>`;

  const table = new Tabulator(container, {
    data: JSON.parse(JSON.stringify(fullyHydratedData)),
    layout: "fitData",
    maxHeight: "400px",
    headerWordWrap: true,
    columnDefaults: {
      minWidth: 30, // in pixels
      headerSort: false,
    },
    columns: [
      { title: "PCE ID", field: "project_code" },
      { title: "Name", field: "sales_reporting_name", maxWidth: 90 },
      { title: "Year", field: "calendar_year", maxWidth: 70 },
      { title: "Month", field: "calendar_month", maxWidth: 75 },
      { title: "Column", field: "column", maxWidth: 170},
      { title: "Orig Value", field: "raw_value" },
      { title: "Carry Fwd", field: "carry_forward" },
      { title: "Annual Avg", field: "annual_average" },
      { title: "Pre/Post Avg", field: "avg_preceding_proceeding" },
      { title: "Same Month Avg", field: "avg_same_month_other_years" , maxWidth: 150},
      { 
        title: "Manual Value", 
        field: "manual_override", 
        editor: "number",
        editorParams: {
          step: "any"
        },
        maxWidth: 100
      },
      {
        title: "Override Action",
        field: "decision",
        editor: "list",
        editorParams: {
          values: [
            "Keep original value",
            "Carry forward",
            "Annual average",
            "Average of preceding and proceeding months",
            "Average of months from other years",
            "Manual override"
          ]
        },
        maxWidth: 150
      },
      {
        title: "Comment",
        field: "comment",
        editor: "input",
        maxWidth: 100
      }
    ]
  });

  // Apply the filter if 'hide decided' toggle checked
  if (hideDecided) {
    table.setFilter(row => {
      return !row.decision || row.decision.trim() === "";
    });
  }

  const updateData = async (cell) => {
    if (cell) {
      const row = cell.getRow();
      const field = cell.getField();
      
      if (field === "manual_override") {
        const val = cell.getValue();
        if (val !== null && val !== "" && !isNaN(val)) {
          await row.update({ decision: "Manual override" });
        }
      }

      const rowData = row.getData();
      await saveReview(rowData); 
      mutable idbTrigger++; 
    }
    
    container.value = table.getData();
    container.dispatchEvent(new CustomEvent("input"));
  };

  table.on("rowMouseEnter", (e, row) => {
    mutable hoveredId = row.getData().identifier;
  });

  table.on("rowMouseLeave", () => {
    mutable hoveredId = null;
  });

  table.on("cellEdited", updateData);
  table.on("tableBuilt", () => updateData());

  container.value = fullyHydratedData;
  return container;
}

// Generate download button
downloadAction = {
  const btn = html`<button class="btn btn-primary my-3"><i class="bi bi-download me-1"></i> Download All Decisions (CSV)</button>`;

  btn.onclick = async () => {
    const savedRecords = await getAllSavedReviews();

    if (!savedRecords || savedRecords.length === 0) {
      alert("No saved overrides found in local storage.");
      return;
    }

    const allKeys = Array.from(new Set(savedRecords.flatMap(Object.keys)));
    const headers = allKeys.join(",");
    
    const rows = savedRecords.map(row =>
      allKeys.map(key => {
        let val = row[key];
        
        if (key === "decision" && val) {
          val = strategyMap.get(val) || val;
        }
        
        let safeVal = String(val ?? '');
        
        // escape double quotes
        safeVal = safeVal.replace(/"/g, '""');
        
        // wrap in quotes
        return `"${safeVal}"`;
      }).join(",")
    );
    
    const csvContent = [headers, ...rows].join("\n");

    // Trigger browser download
    const blob = new Blob([csvContent], { type: "text/csv;charset=utf-8;" });
    const url = URL.createObjectURL(blob);
    const link = document.createElement("a");
    link.setAttribute("href", url);
    link.setAttribute("download", `outlier_reviews_${new Date().toISOString().slice(0, 10)}.csv`);
    document.body.appendChild(link);
    link.click();
    document.body.removeChild(link);
  };

  return btn;
}
